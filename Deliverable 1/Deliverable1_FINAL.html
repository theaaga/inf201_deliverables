Task 4: Heatwaves¶

Look for heatwaves in Ås in 2025. In Norway a heatwave is defined as a consecutive period of five or more day where the max temperature is >= 27 degrees Celcius.

Write a function that count the number of heatwaves in a year.

  • The function should write a summary to a yml file, formatted as the example file.
  • Write the function so that it will work with any station name/id and year
In [49]:
def get_max_temperature_data(station_id, year, client_id):
    """Get max temperature from a given station for a given year"""

    # Retrieve data
    endpoint = 'https://frost.met.no/observations/v0.jsonld'
    parameters = {
        'sources': station_id,
        'elements': 'max(air_temperature P1D)',
        'referencetime': f'{year}-01-01/{year}-12-31',
    }
    
    response = requests.get(endpoint, params=parameters, auth=(client_id, ''))

    # Check that retrieval was successful, if yes, process into a dataframe
    if response.status_code == 200:
        data = response.json()['data']
        rows = []
        for entry in data:
            date = entry['referenceTime']
            # Check that the observation exists
            if entry['observations']:
                value = entry['observations'][0]['value']
                rows.append({'date': date, 'max_temp': value})
            
        df = pd.DataFrame(rows)
        # Convert date to datetime-object and sort dataframe
        df['date'] = pd.to_datetime(df['date'])
        df = df.sort_values('date').reset_index(drop=True)
        return df
    # If unsuccesful, raise error
    else:
        print(f'Error retrieving data: {response.status_code} - {response.text}')
        return None


def detect_heatwaves(station_name, station_id, year, client_id, filename='heatwave_summary.yaml'):
    """
    Counts heatwaves (5+ days of max temperature >= 27°C), 
    Writes summary to yml-file. 
    """

    df = get_max_temperature_data(station_id, year, client_id)
    if df is None or df.empty:
        print("No data found")
        return None
    
    # Find days that are above 27 degress
    df['above_27'] = df['max_temp'] >= 27

    # Iterate through rows and find consecutive days 
    heatwave_ranges = []
    current_streak = []
    
    for _, row in df.iterrows():
        if row['above_27']:
            current_streak.append(row)
        else:
            if len(current_streak) >= 5:
                start_date = str(current_streak[0]['date'].date())
                end_date = str(current_streak[-1]['date'].date())
                heatwave_ranges.append(f"{start_date} - {end_date}")
            current_streak = []
            
    if len(current_streak) >= 5:
        start_date = str(current_streak[0]['date'].date())
        end_date = str(current_streak[-1]['date'].date())
        heatwave_ranges.append(f"{start_date} - {end_date}")
        
    summary = {
        'dates': heatwave_ranges,
        'station_id': station_id,
        'station_name': station_name,
        'year': year
    }
    
    # Write result to a .yml-file
    with open(filename, 'w', encoding='utf-8') as f:
        yaml.dump(summary, f, allow_unicode=True, sort_keys=False)

    return summary

Write a function that calculates the 7-day moving average for temperature

In [51]:
def calculate_moving_average(df, window_size=7, value_column='value'):
    """
    Calculates moving average for given column in a dataframe
    Standard window-size is 7 days. 
    """
    df_ma = df.copy()
    # Use pandas rolling-function to calculate moving average 
    df_ma['moving_average'] = df_ma[value_column].rolling(window=window_size).mean()
    return df_ma
